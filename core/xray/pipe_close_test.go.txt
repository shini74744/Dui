package singbridge

import (
	"context"
	"io"
	"net"
	"testing"
	"time"

	"github.com/xtls/xray-core/common/buf"
	"github.com/xtls/xray-core/transport"
	"github.com/xtls/xray-core/transport/pipe"
)

func TestCloseWriteDrainsRequestAndPreservesResponse(t *testing.T) {
	upR, upW := pipe.New()
	downR, downW := pipe.New()
	w := &PipeConnWrapper{R: &buf.BufferedReader{Reader: downR}, W: upW}
	defer w.Close()
	defer downW.Close()
	if _, err := w.Write([]byte("request")); err != nil {
		t.Fatal(err)
	}
	if err := w.CloseWrite(); err != nil {
		t.Fatal(err)
	}
	request, err := io.ReadAll(&buf.BufferedReader{Reader: upR})
	if err != nil || string(request) != "request" {
		t.Fatalf("request=%q err=%v", request, err)
	}
	if err := downW.WriteMultiBuffer(buf.MultiBuffer{buf.FromBytes([]byte("response"))}); err != nil {
		t.Fatal(err)
	}
	response := make([]byte, 8)
	if _, err := io.ReadFull(w, response); err != nil || string(response) != "response" {
		t.Fatalf("response=%q err=%v", response, err)
	}
}

func TestCloseUnblocksPendingRead(t *testing.T) {
	upR, upW := pipe.New()
	downR, downW := pipe.New()
	defer upR.Interrupt()
	defer downW.Close()
	w := &PipeConnWrapper{R: &buf.BufferedReader{Reader: downR}, W: upW}
	done := make(chan error, 1)
	go func() { _, err := w.Read(make([]byte, 1)); done <- err }()
	if err := w.Close(); err != nil {
		t.Fatal(err)
	}
	select {
	case err := <-done:
		if err == nil {
			t.Fatal("read unexpectedly succeeded")
		}
	case <-time.After(time.Second):
		t.Fatal("read remained blocked after Close")
	}
}

func TestCopyConnPropagatesClientEOF(t *testing.T) {
	client, server := net.Pipe()
	defer client.Close()
	defer server.Close()
	upR, upW := pipe.New()
	downR, downW := pipe.New()
	defer upR.Interrupt()
	defer downR.Interrupt()
	defer downW.Close()
	ctx, cancel := context.WithTimeout(context.Background(), 2*time.Second)
	defer cancel()
	copied := make(chan error, 1)
	go func() { copied <- CopyConn(ctx, server, &transport.Link{Reader: downR, Writer: upW}, server) }()
	client.Close()
	eof := make(chan error, 1)
	go func() { _, err := upR.ReadMultiBuffer(); eof <- err }()
	select {
	case err := <-eof:
		if err != io.EOF {
			t.Fatalf("expected EOF, got %v", err)
		}
	case <-time.After(time.Second):
		t.Fatal("client EOF was not propagated")
	}
	downW.Close()
	select {
	case <-copied:
	case <-time.After(time.Second):
		t.Fatal("copy did not finish")
	}
}
